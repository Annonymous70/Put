    function updateCloudStatus(status, isError = false) {

    async function loadGame() {
        updateCloudStatus("Menyelaraskan cloud...");
        try {
            let response = await fetch(API_URL);
            if (response.ok) {
                let parsed = await response.json();
                if (parsed && parsed.users) {
                    db.users = parsed.users || {};
                    db.companies = parsed.companies || {};
                    db.marketProducts = parsed.marketProducts || [];
                    db.loans = parsed.loans || [];

                    db.adminCode = parsed.adminCode || "Bisnis2026";

                    if (currentUser && db.users[currentUser.name]) {
                        currentUser = db.users[currentUser.name];
                        let walletEl = document.getElementById('ui-wallet-badge');
                        if (walletEl) walletEl.innerText = `Saldo: Rp ${currentUser.wallet.toLocaleString('id-ID')}`;
                    }
                }
                updateCloudStatus("☁️ Terhubung & Sinkron");
            }
        } catch (e) {
            console.error("Gagal load:", e);
            updateCloudStatus("⚠️ Gagal Terhubung ke Cloud", true);
        }
        renderDebugUsers();
    }

    async function saveGame() {
        updateCloudStatus("Menyimpan ke cloud...");
        try {
            const saveData = {
                users: db.users,
                companies: db.companies,
                marketProducts: db.marketProducts,
                loans: db.loans,
                dividendProposals: db.dividendProposals,
                adminCode: db.adminCode
            };
            let res = await fetch(API_URL, {
                method: 'PUT',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(saveData)
            });
            if (!res.ok) throw new Error("Gagal menyimpan server");
            updateCloudStatus("☁️ Perubahan Tersimpan Online");
        } catch (e) {
            console.error("Gagal save:", e);
            updateCloudStatus("❌ Gagal Menyimpan", true);
        }
    }

    async function backgroundSync() {
        try {
            let response = await fetch(API_URL);
            if (response.ok) {
                let parsed = await response.json();
                if (parsed && parsed.users) {
                    db.users = parsed.users || {};
                    db.companies = parsed.companies || {};
                    db.marketProducts = parsed.marketProducts || [];
                    db.loans = parsed.loans || [];
                    db.dividendProposals = parsed.dividendProposals || [];
                    renderDebugUsers();
                    if (currentUser && !document.getElementById('screen-dashboard').classList.contains('hidden')) {
                        currentUser = db.users[currentUser.name] || currentUser;
                        updateDayDisplay();
                    }
                }
            }
        } catch (e) {}
    }

    async function handleAuthSubmit() {
        const name = document.getElementById('input-name').value.trim();
        const password = document.getElementById('input-password').value.trim();
        if (!name || !password) return alert("Isi nama dan password!");
        await loadGame();

        if (authMode === 'reg') {
            if (!selectedRole) return alert("Pilih role!");
            if (db.users[name]) return alert("Nama akun sudah terdaftar di cloud!");
            
            db.users[name] = { 
                name: name, password: password, role: selectedRole, 
                companyId: null, certified: false, wallet: 1000000, 
                producedToday: 0, lastDayProcessed: 1, portfolio: {}
            };
            currentUser = db.users[name];
            await saveGame();
            openDashboard();
        } else {
            let u = db.users[name];
            if (!u) return alert("Akun tidak ditemukan di Cloud!");
            if (u.password !== password) return alert("Password salah!");
            if (u.role === 'Admin') {
                let code = prompt("Kode Akses Admin:");
                if (code !== db.adminCode) return alert("Kode admin salah!");
            }
            currentUser = u;
            await saveGame();
            openDashboard();
        }
    }

    function updateDayDisplay() {
        let currentDay = 1;
        if (currentUser && currentUser.companyId && db.companies[currentUser.companyId]) {
            currentDay = db.companies[currentUser.companyId].day;
        } else {
            for (let id in db.companies) {
                let comp = db.companies[id];
                if (comp.staff.some(s => s.name === currentUser.name)) {
                    currentDay = comp.day;
                    break;
                }
            }
        }
        let dayEl = document.getElementById('ui-day-badge');
        if (dayEl) dayEl.innerText = `Hari ke-${currentDay}`;
    }

    async function processLoan(idx, status) {
        let l = db.loans[idx];
        l.status = status;

            l.totalDue = l.amount;
            if (l.targetType === 'Personal') db.users[l.applicantName].wallet += l.amount;

        renderAdmin();
    }

    function getMyCompany() {
        for (let id in db.companies) if (db.companies[id].ceoName === currentUser.name) return db.companies[id];

    function getCompanyValuation(comp) {
        let val = comp.treasury;
        comp.products.forEach(p => val += (p.stock || 0) * (p.price || p.cost || 0));
        return val;
    }

    async function ceoAdvanceDay() {
        let myComp = getMyCompany();
        let totalSal = 0;
        myComp.staff.forEach(st => totalSal += (myComp.salaries[st.role] || 0));
        if (myComp.treasury < totalSal) return alert("Kas PT tidak cukup bayar gaji!");
        myComp.treasury -= totalSal;
        myComp.day += 1;
        myComp.staff.forEach(st => {
            let u = db.users[st.name];
            if (u) { u.wallet += (myComp.salaries[st.role] || 0); u.producedToday = 0; }
        });
        await saveGame();
        alert(`Hari ke-${myComp.day}! Gaji dibayarkan.`);

        updateDayDisplay();
    }

        db.companies[currentUser.companyId].gmProposals.push({ gmName: currentUser.name, name: name, cost: cost });
        await saveGame();
        alert("Usulan dikirim ke CEO.");
    }

    function renderFinanceTable(comp) {
        const tbody = document.getElementById('table-finance-products');
        tbody.innerHTML = "";
        comp.products.forEach(p => {
            let margin = p.price > 0 ? (((p.price - p.cost) / p.cost) * 100).toFixed(1) + "%" : "0%";
            tbody.innerHTML += `<tr><td>${p.name}</td><td>Rp ${p.cost}</td><td><input type="number" id="price-${p.id}" value="${p.price}" style="width:100px;margin:0;" /></td><td style="color:var(--success);">${margin}</td><td><button class="btn btn-success" onclick="savePrice('${comp.id}', ${p.id})">Simpan</button></td></tr>`;
        });
    }

    async function savePrice(compId, prodId) {
        let comp = db.companies[compId];
        let p = comp.products.find(x => x.id === prodId);
        p.price = parseFloat(document.getElementById(`price-${prodId}`).value);
        await saveGame();
        alert("Harga disimpan.");
        renderFinanceTable(comp);
    }

    function renderMarketingWorkspace(comp) {
        const buyerSelect = document.getElementById('direct-sale-buyer');
        buyerSelect.innerHTML = '<option value="">-- Pilih --</option>';
        for (let name in db.users) if (name !== currentUser.name) buyerSelect.innerHTML += `<option value="${name}">${name}</option>`;

        const prodSelect = document.getElementById('direct-sale-product');
        prodSelect.innerHTML = '<option value="">-- Pilih --</option>';
        comp.products.forEach(p => prodSelect.innerHTML += `<option value="${p.id}">${p.name} (Stok: ${p.stock || 0})</option>`);

        const tbody = document.getElementById('table-marketing-products');
        tbody.innerHTML = "";
        comp.products.forEach(p => {
            let pub = db.marketProducts.some(m => m.prodId === p.id);
            let act = pub ? `<button class="btn btn-danger" style="padding:4px;" onclick="unpublishMarket(${p.id})">Tarik</button>` : `<button class="btn btn-primary" style="padding:4px;" onclick="publishMarket('${comp.id}', ${p.id})">Publish</button>`;
            tbody.innerHTML += `<tr><td>${p.name}</td><td>Rp ${p.price}</td><td><strong>${p.stock || 0}</strong></td><td>${act}</td></tr>`;
        });
    }

    async function executeDirectSale() {
        let buyerName = document.getElementById('direct-sale-buyer').value;
        let prodId = parseInt(document.getElementById('direct-sale-product').value);
        let qty = parseInt(document.getElementById('direct-sale-qty').value);
        let comp = db.companies[currentUser.companyId];
        let p = comp.products.find(x => x.id === prodId);
        let buyer = db.users[buyerName];
        let total = p.price * qty;
        if (buyer.wallet < total) return alert("Saldo pembeli kurang!");
        buyer.wallet -= total;
        p.stock -= qty;
        comp.treasury += total;
        await saveGame();
        alert("Penjualan sukses!");
        renderMarketingWorkspace(comp);
    }

    async function publishMarket(compId, prodId) {
        let comp = db.companies[compId];
        let p = comp.products.find(x => x.id === prodId);
        db.marketProducts.push({ compId: comp.id, compName: comp.name, prodId: p.id, name: p.name, price: p.price });
        await saveGame();
        alert("Dipublikasikan!");
        renderMarketingWorkspace(comp);
    }

    async function unpublishMarket(prodId) {
        db.marketProducts = db.marketProducts.filter(m => m.prodId !== prodId);
        await saveGame();
        renderMarketingWorkspace(db.companies[currentUser.companyId]);
    }

    function renderProductionWorkspace(comp) {
        const statusEl = document.getElementById('ui-cert-status');
        if (currentUser.certified) {
            statusEl.innerText = "Bersertifikasi (Maks 100 Unit/Hari)";
            statusEl.style.color = "var(--success)";
        }
        let limit = currentUser.certified ? 100 : 20;
        let sisa = limit - currentUser.producedToday;
        const tbody = document.getElementById('table-production-products');
        tbody.innerHTML = "";
        comp.products.forEach(p => {
            tbody.innerHTML += `<tr><td>${p.name}</td><td>Rp ${p.price || 0}</td><td>Rp ${p.cost}</td><td><strong>${p.stock || 0}</strong></td><td><strong>${sisa}</strong></td><td><button class="btn btn-primary" style="padding:4px;" onclick="produceProduct('${comp.id}', ${p.id})">Produksi 10</button></td></tr>`;
        });
    }

    async function takeCertification() {
        if (currentUser.wallet < 50000) return alert("Saldo kurang!");
        currentUser.wallet -= 50000;
        currentUser.certified = true;
        await saveGame();
        alert("Sertifikasi sukses!");
        renderWorkspace();
    }

    async function produceProduct(compId, prodId) {
        let limit = currentUser.certified ? 100 : 20;
        if (currentUser.producedToday + 10 > limit) return alert("Batas kuota habis!");
        let comp = db.companies[compId];
        let p = comp.products.find(x => x.id === prodId);
        let hpp = 10 * p.cost;
        if (comp.treasury < hpp) return alert("Kas PT kurang!");
        comp.treasury -= hpp;
        currentUser.producedToday += 10;
        p.stock = (p.stock || 0) + 10;
        await saveGame();
        alert("Produksi 10 unit sukses!");
        renderProductionWorkspace(comp);
    }

    // --- MARKETPLACE & OTHER TABS ---
    function renderMarketplace() {
        const c = document.getElementById('marketplace-container');
        c.innerHTML = "";
        db.marketProducts.forEach(m => {
            c.innerHTML += `<div class="market-item"><h4>${m.name}</h4><p>PT: ${m.compName}</p><p style="color:var(--success);font-weight:bold;">Rp ${m.price.toLocaleString('id-ID')}</p><button class="btn btn-success" style="width:100%;margin-top:8px;" onclick="buyMarketProduct('${m.compId}', ${m.prodId}, ${m.price})">Beli</button></div>`;
        });
        if (db.marketProducts.length === 0) c.innerHTML = `<p style="color:var(--secondary)">Marketplace kosong.</p>`;
    }

    async function buyMarketProduct(compId, prodId, price) {
        if (currentUser.wallet < price) return alert("Saldo kurang!");
        let comp = db.companies[compId];
        let prod = comp.products.find(x => x.id === prodId);
        prod.stock -= 1;
        currentUser.wallet -= price;
        comp.treasury += price;
        await saveGame();
        alert("Pembelian berhasil!");
        renderMarketplace();
    }

    function renderStockExchange() {
        const tm = document.getElementById('stock-market-table');
        tm.innerHTML = "";
        for (let id in db.companies) {
            let c = db.companies[id];
            if (!c.isPublic) continue;
            tm.innerHTML += `<tr><td>${c.name}</td><td>Rp ${c.stockPrice}</td><td>${c.totalShares}</td><td>${c.availableShares}</td><td>Rp ${c.totalShares * c.stockPrice}</td><td><button class="btn btn-primary" onclick="buyShares('${c.id}')">Beli</button></td></tr>`;
        }
        let myComp = getMyCompany();
        if (myComp && myComp.isPublic) document.getElementById('ceo-dividend-creator').classList.remove('hidden');
    }

    async function buyShares(compId) {
        let comp = db.companies[compId];
        let qty = parseInt(prompt("Lembar saham:", "100"));
        let cost = qty * comp.stockPrice;
        if (currentUser.wallet < cost) return alert("Saldo kurang!");
        currentUser.wallet -= cost;
        comp.availableShares -= qty;
        currentUser.portfolio[compId] = (currentUser.portfolio[compId] || 0) + qty;
        await saveGame();
        alert("Beli saham sukses!");
        renderStockExchange();
    }
